import express from "express"; import cors from "cors"; import OpenAI from "openai"; const app = express(); const PORT = process.env.PORT || 10000; const MODEL = process.env.OPENAI_MODEL || "gpt-5.6-luna"; app.use(cors()); app.use(express.json({ limit: "1mb" })); // -------------------------------------------------- // OPENAI // -------------------------------------------------- const apiKey = process.env.OPENAI_API_KEY; const openai = apiKey ? new OpenAI({ apiKey }) : null; // -------------------------------------------------- // HEALTH CHECK // -------------------------------------------------- app.get("/", (req, res) => { res.json({ ok: true, name: "ViralForge AI Studio Backend", status: openai ? "AI connected" : "AI key missing", model: MODEL }); }); app.get("/health", (req, res) => { res.json({ ok: true, openaiConfigured: Boolean(apiKey), model: MODEL }); }); // -------------------------------------------------- // HELPER // -------------------------------------------------- async function generateAI(prompt) { if (!openai) { throw new Error( "OPENAI_API_KEY is not configured on the server." ); } const response = await openai.responses.create({ model: MODEL, input: prompt }); return response.output_text || ""; } // -------------------------------------------------- // MAIN AI ENDPOINT // -------------------------------------------------- app.post("/api/generate", async (req, res) => { try { const { type = "script", topic = "", niche = "General", platform = "TikTok", style = "High Energy", length = "30 seconds", audience = "Everyone", tone = "Confident", goal = "Growth", count = 3, text = "" } = req.body || {}; if (!openai) { return res.status(503).json({ ok: false, error: "AI is not configured yet. Add OPENAI_API_KEY to your Render environment variables." }); } // ---------------------------------------------- // SCRIPTS // ---------------------------------------------- if (type === "script" || type === "scripts") { const prompt = ` You are ViralForge, an expert short-form content strategist. Create ${Math.min(Number(count) || 3, 5)} DIFFERENT short-form video scripts. Topic: ${topic || "Create a strong topic yourself"} Niche: ${niche} Platform: ${platform} Style: ${style} Length: ${length} Audience: ${audience} Requirements: - Make each version genuinely different. - Strong hook in the first sentence. - Natural human wording. - Do not sound like generic AI writing. - Keep sentences easy to say out loud. - Include a clear payoff. - Include a natural CTA. - Do not guarantee virality. - Do not use fake statistics. - Make the content useful or entertaining. Format: VERSION 1 HOOK: ... SCRIPT: ... CTA: ... VERSION 2 HOOK: ... SCRIPT: ... CTA: ... Continue for all requested versions. `; const result = await generateAI(prompt); return res.json({ ok: true, type: "scripts", result }); } // ---------------------------------------------- // HOOKS // ---------------------------------------------- if (type === "hooks" || type === "hook") { const prompt = ` You are ViralForge Hook Lab. Generate 12 unique short-form video hooks. Topic: ${topic || "a useful creator topic"} Niche: ${niche} Hook style: ${style} Platform: ${platform} Requirements: - Each hook must be different. - Make them sound like something a real creator would say. - Avoid repetitive formulas. - Avoid fake claims. - Use curiosity, tension, specificity, questions, surprising observations, or direct statements. - Keep each hook concise. - Do not guarantee views. Return exactly 12 numbered hooks. `; const result = await generateAI(prompt); return res.json({ ok: true, type: "hooks", result }); } // ---------------------------------------------- // IDEAS // ---------------------------------------------- if (type === "ideas" || type === "idea") { const prompt = ` You are ViralForge Idea Lab. Generate 20 original short-form content ideas. Niche: ${niche} Platform: ${platform} Goal: ${goal} Requirements: - Every idea should be meaningfully different. - Give each idea a strong title. - Include the angle. - Include why someone would want to watch. - Avoid fake statistics. - Avoid generic ideas whenever possible. - Make ideas realistic for a creator to make. Format: 1. TITLE Angle: Why it could work: Continue through 20. `; const result = await generateAI(prompt); return res.json({ ok: true, type: "ideas", result }); } // ---------------------------------------------- // CAPTION // ---------------------------------------------- if (type === "caption" || type === "captions") { const prompt = ` You are ViralForge Caption Maker. Create a strong social-media caption. Topic: ${topic || "this video"} Tone: ${tone} Platform: ${platform} Requirements: - Natural human writing. - Do not sound corporate. - Start with an interesting first line. - Match the requested tone. - Encourage comments naturally. - Include a small number of relevant hashtags. - Do not promise virality. - Do not use fake statistics. Return: CAPTION: ... HASHTAGS: ... `; const result = await generateAI(prompt); return res.json({ ok: true, type: "caption", result }); } // ---------------------------------------------- // VIRAL SCORE // ---------------------------------------------- if (type === "score" || type === "analyze") { const prompt = ` You are ViralForge's content scoring system. Analyze this short-form content: ${text || topic} Give a writing-quality score from 0-100. Evaluate: 1. Hook strength 2. Curiosity 3. Clarity 4. Specificity 5. Retention potential 6. Structure 7. CTA Important: - This is NOT a prediction of actual views. - Do not claim you can guarantee virality. - Give practical improvements. Return: SCORE: X/100 HOOK: ... CURIOSITY: ... CLARITY: ... SPECIFICITY: ... RETENTION: ... STRUCTURE: ... CTA: ... TOP 3 IMPROVEMENTS: 1. 2. 3. `; const result = await generateAI(prompt); return res.json({ ok: true, type: "score", result }); } // ---------------------------------------------- // UNKNOWN TYPE // ---------------------------------------------- return res.status(400).json({ ok: false, error: "Unknown generation type. Use script, hooks, ideas, caption, or score." }); } catch (error) { console.error("ViralForge AI error:", error); const status = error?.status && Number.isInteger(error.status) ? error.status : 500; return res.status(status).json({ ok: false, error: error?.message || "Something went wrong while generating content." }); } }); // -------------------------------------------------- // ERROR HANDLER // -------------------------------------------------- app.use((err, req, res, next) => { console.error("Server error:", err); res.status(500).json({ ok: false, error: "Internal server error." }); }); // -------------------------------------------------- // START SERVER // -------------------------------------------------- app.listen(PORT, "0.0.0.0", () => { console.log(""); console.log("===================================="); console.log(" VIRALFORGE AI BACKEND"); console.log("===================================="); console.log(`Port: ${PORT}`); console.log(`OpenAI: ${openai ? "CONNECTED" : "NOT CONFIGURED"}`); console.log(`Model: ${MODEL}`); console.log("===================================="); console.log(""); });
